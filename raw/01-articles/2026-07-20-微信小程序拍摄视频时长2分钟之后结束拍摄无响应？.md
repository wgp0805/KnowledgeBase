# 微信小程序拍摄视频时长2分钟之后结束拍摄无响应？

微信小程序拍摄视频时长2分钟之后结束拍摄无响应
    
    
    <script setup lang="ts">
    import { useToast } from '@/wd'
    
    interface CameraFileResult {
      errMsg?: string
      tempThumbPath?: string
      tempVideoPath?: string
    }
    
    interface CameraCallbackResult {
      errMsg?: string
    }
    
    interface NativeStartRecordOptions {
      /**
       * 微信公开类型暂未声明该参数，这里按当前业务相机的调用方式透传秒数，
       * 设为 5 分钟是为了留出 2 分钟以上的手动停止测试窗口。
       */
      timeout: number
      timeoutCallback: (result: CameraFileResult) => void
      success: (result: CameraCallbackResult) => void
      fail: (result: CameraCallbackResult) => void
      complete: (result: CameraCallbackResult) => void
    }
    
    interface NativeStopRecordOptions {
      success: (result: CameraFileResult) => void
      fail: (result: CameraCallbackResult) => void
      complete: (result: CameraCallbackResult) => void
    }
    
    interface NativeCameraContext {
      startRecord: (options: NativeStartRecordOptions) => void
      stopRecord: (options: NativeStopRecordOptions) => void
    }
    
    interface CallbackLog {
      id: number
      name: string
      time: string
      detail: string
    }
    
    type TestStatus = 'idle' | 'ready' | 'recording' | 'stopping' | 'timeout' | 'stopped' | 'error'
    
    const CAMERA_ID = 'nativeCameraDurationTest'
    const RECORD_TIMEOUT_SECONDS = 300
    
    const toast = useToast()
    const cameraContext = shallowRef<NativeCameraContext | null>(null)
    const cameraVisible = ref(false)
    const cameraReady = ref(false)
    const isStarting = ref(false)
    const isRecording = ref(false)
    const isStopping = ref(false)
    const elapsedSeconds = ref(0)
    const stopSuccessTriggered = ref(false)
    const stopFailTriggered = ref(false)
    const stopCompleteTriggered = ref(false)
    const testStatus = ref<TestStatus>('idle')
    const callbackLogs = ref<CallbackLog[]>([])
    
    let recordStartedAt = 0
    let recordTimer: ReturnType<typeof setInterval> | null = null
    let logSequence = 0
    
    const statusText = computed(() => {
      const statusMap: Record<TestStatus, string> = {
        idle: '等待打开相机',
        ready: '相机已就绪',
        recording: '正在录制',
        stopping: '正在等待 stopRecord 回调',
        timeout: '录制被 timeoutCallback 结束',
        stopped: '手动结束已返回',
        error: '测试发生错误',
      }
      return statusMap[testStatus.value]
    })
    
    const statusClass = computed(() => `test-status--${testStatus.value}`)
    const elapsedText = computed(() => formatDuration(elapsedSeconds.value))
    const canManuallyStop = computed(() => isRecording.value && !isStopping.value)
    
    function formatDuration(seconds: number) {
      const safeSeconds = Math.max(0, Math.floor(seconds))
      const minutes = Math.floor(safeSeconds / 60)
      const remainingSeconds = safeSeconds % 60
      return `${String(minutes).padStart(2, '0')}:${String(remainingSeconds).padStart(2, '0')}`
    }
    
    function formatClockTime(date = new Date()) {
      return [date.getHours(), date.getMinutes(), date.getSeconds()]
        .map((value) => String(value).padStart(2, '0'))
        .join(':')
    }
    
    function stringifyResult(result: CameraFileResult | CameraCallbackResult) {
      const parts = [
        result.errMsg,
        'tempVideoPath' in result && result.tempVideoPath
          ? `tempVideoPath: ${result.tempVideoPath}`
          : '',
        'tempThumbPath' in result && result.tempThumbPath
          ? `tempThumbPath: ${result.tempThumbPath}`
          : '',
      ].filter(Boolean)
      return parts.join('\n') || '无返回内容'
    }
    
    function addLog(name: string, result: CameraFileResult | CameraCallbackResult = {}) {
      callbackLogs.value.unshift({
        id: ++logSequence,
        name,
        time: formatClockTime(),
        detail: stringifyResult(result),
      })
    }
    
    function clearRecordTimer() {
      if (recordTimer) {
        clearInterval(recordTimer)
        recordTimer = null
      }
    }
    
    function startRecordTimer() {
      clearRecordTimer()
      recordStartedAt = Date.now()
      elapsedSeconds.value = 0
      recordTimer = setInterval(() => {
        elapsedSeconds.value = Math.floor((Date.now() - recordStartedAt) / 1000)
      }, 1000)
    }
    
    function finishRecord(status: Extract<TestStatus, 'timeout' | 'stopped' | 'error'>) {
      if (recordStartedAt > 0) {
        elapsedSeconds.value = Math.floor((Date.now() - recordStartedAt) / 1000)
      }
      clearRecordTimer()
      isStarting.value = false
      isRecording.value = false
      isStopping.value = false
      recordStartedAt = 0
      testStatus.value = status
    }
    
    async function openCamera() {
      cameraVisible.value = true
      cameraReady.value = false
      stopSuccessTriggered.value = false
      stopFailTriggered.value = false
      stopCompleteTriggered.value = false
      testStatus.value = 'idle'
      callbackLogs.value = []
      await nextTick()
      addLog('已挂载原生 camera 组件')
    }
    
    function closeCamera() {
      if (isRecording.value || isStarting.value) {
        toast.warning('请先结束当前录制')
        return
      }
      clearRecordTimer()
      cameraContext.value = null
      cameraReady.value = false
      cameraVisible.value = false
      testStatus.value = 'idle'
    }
    
    function handleCameraInitDone() {
      try {
        cameraContext.value = (
          uni.createCameraContext as unknown as (id?: string) => NativeCameraContext
        )(CAMERA_ID)
        cameraReady.value = Boolean(cameraContext.value)
        testStatus.value = cameraReady.value ? 'ready' : 'error'
        addLog(cameraReady.value ? 'camera initdone' : '创建 CameraContext 失败')
      } catch (error) {
        cameraContext.value = null
        cameraReady.value = false
        testStatus.value = 'error'
        addLog('创建 CameraContext 异常', { errMsg: String(error) })
      }
    }
    
    function handleCameraError(event: any) {
      finishRecord('error')
      cameraReady.value = false
      addLog('camera error', event?.detail || event || {})
    }
    
    function handleCameraStop(event: any) {
      addLog('camera stop 事件', event?.detail || event || {})
    }
    
    function handleTimeout(result: CameraFileResult) {
      finishRecord('timeout')
      addLog(`timeoutCallback 提前结束（实际 ${elapsedText.value}）`, result)
      toast.warning(`录制在 ${elapsedText.value} 被 timeoutCallback 结束`)
    }
    
    function startRecord() {
      if (!cameraReady.value || !cameraContext.value || isStarting.value || isRecording.value) {
        toast.warning('相机尚未就绪')
        return
      }
    
      stopSuccessTriggered.value = false
      stopFailTriggered.value = false
      stopCompleteTriggered.value = false
      isStarting.value = true
      callbackLogs.value = []
      addLog('调用 startRecord', { errMsg: `timeout: ${RECORD_TIMEOUT_SECONDS} 秒` })
    
      try {
        cameraContext.value.startRecord({
          timeout: RECORD_TIMEOUT_SECONDS,
          timeoutCallback: handleTimeout,
          success: (result) => {
            isStarting.value = false
            isRecording.value = true
            testStatus.value = 'recording'
            startRecordTimer()
            addLog('startRecord success', result)
          },
          fail: (result) => {
            finishRecord('error')
            addLog('startRecord fail', result)
          },
          complete: (result) => {
            addLog('startRecord complete（仅表示调用完成）', result)
          },
        })
      } catch (error) {
        finishRecord('error')
        addLog('startRecord 调用异常', { errMsg: String(error) })
      }
    }
    
    function stopRecord() {
      if (!cameraContext.value || !canManuallyStop.value) return
      isStopping.value = true
      testStatus.value = 'stopping'
      addLog(`调用 stopRecord（录制 ${elapsedText.value}）`)
      cameraContext.value.stopRecord({
        success: (result) => {
          console.log('原生相机长视频 stopRecord success 回调已触发', result)
          stopSuccessTriggered.value = true
          finishRecord('stopped')
          addLog(`stopRecord success（录制 ${elapsedText.value}）`, result)
        },
        fail: (result) => {
          console.error('原生相机长视频 stopRecord fail 回调已触发', result)
          stopFailTriggered.value = true
          finishRecord('error')
          addLog('stopRecord fail', result)
        },
        complete: (result) => {
          console.log('原生相机长视频 stopRecord complete 回调已触发', result)
          stopCompleteTriggered.value = true
          if (!stopSuccessTriggered.value && !stopFailTriggered.value) {
            finishRecord('error')
          }
          addLog('stopRecord complete', result)
        },
      })
    }
    
    onUnmounted(() => {
      clearRecordTimer()
    })
    </script>
    
    <template>
      <view class="native-camera-test">
        <view class="test-header">
          <view class="test-title-group">
            <text class="test-title">微信原生 Camera 长视频停止测试</text>
            <text class="test-description">
              可随时手动调用 stopRecord；录制 2 分钟以上再结束时，可重点验证长视频回调。
            </text>
          </view>
          <wd-tag type="warning" plain>测试组件</wd-tag>
        </view>
    
        <!-- #ifdef MP-WEIXIN -->
        <template v-if="cameraVisible">
          <camera
            :id="CAMERA_ID"
            class="camera-preview"
            mode="normal"
            resolution="low"
            device-position="back"
            flash="off"
            @initdone="handleCameraInitDone"
            @error="handleCameraError"
            @stop="handleCameraStop"
          />
    
          <view class="test-metrics">
            <view class="test-metric">
              <text class="test-metric-label">当前状态</text>
              <text class="test-status" :class="statusClass">{{ statusText }}</text>
            </view>
            <view class="test-metric">
              <text class="test-metric-label">当前录制时长</text>
              <text class="test-duration">{{ elapsedText }}</text>

            <view class="test-metric">
              <text class="test-metric-label">success 回调</text>
              <text
                class="test-callback-result"
                :class="{ 'test-callback-result--success': stopSuccessTriggered }"
              >
                {{ stopSuccessTriggered ? '已触发' : '未触发' }}
              </text>
            </view>
            <view class="test-metric">
              <text class="test-metric-label">fail 回调</text>
              <text
                class="test-callback-result"
                :class="{ 'test-callback-result--error': stopFailTriggered }"
              >
                {{ stopFailTriggered ? '已触发' : '未触发' }}
              </text>
            </view>
            <view class="test-metric">
              <text class="test-metric-label">complete 回调</text>
              <text
                class="test-callback-result"
                :class="{ 'test-callback-result--success': stopCompleteTriggered }"
              >
                {{ stopCompleteTriggered ? '已触发' : '未触发' }}
              </text>
            </view>
          </view>
    
          <view class="test-actions">
            <wd-button
              type="primary"
              block
              :loading="isStarting"
              :disabled="!cameraReady || isRecording"
              @click="startRecord"
            >
              {{ isRecording ? '录制中…' : '开始长视频录制' }}
            </wd-button>
            <wd-button
              v-if="isRecording"
              type="error"
              plain
              block
              :loading="isStopping"
              :disabled="!canManuallyStop"
              @click="stopRecord"
            >
              手动结束并测试回调
            </wd-button>
            <wd-button v-else type="default" plain block @click="closeCamera">收起测试相机</wd-button>
          </view>
    
          <view v-if="callbackLogs.length" class="test-logs">
            <text class="test-logs-title">回调日志（最新在前）</text>
            <view v-for="log in callbackLogs" :key="log.id" class="test-log-item">
              <view class="test-log-header">
                <text class="test-log-name">{{ log.name }}</text>
                <text class="test-log-time">{{ log.time }}</text>
              </view>
              <text class="test-log-detail">{{ log.detail }}</text>
            </view>
          </view>
        </template>
    
        <wd-button v-else type="primary" plain block @click="openCamera">打开测试相机</wd-button>
        <!-- #endif -->

        <view class="platform-tip">此 Demo 只在微信小程序端编译，请使用微信开发者工具或真机测试。</view>
        <!-- #endif -->
      </view>
    </template>
